import type { PixelSprite } from '../objects/types';

// 原寸32×32。既存の正面素材の王冠・白髪・赤いマント・金の縁を引き継ぐ。
// .＝透明、0＝輪郭、1〜3＝赤、4〜6＝金、7〜9＝白髪／白い縁、a〜c＝肌。
const palette = [
  '#211e29', '#651529', '#a71d36', '#d33243',
  '#98591f', '#d69a36', '#f8cf63', '#64616f', '#babac6', '#f5f1e5',
  '#b97846', '#ebad67', '#ffcf89', '#563423', '#86603a', '#ede0b0',
];

function paint(pose: 'back' | 'right'): PixelSprite {
  const pixels = Array.from({ length: 32 }, () => Array<string>(32).fill('.'));
  const rect = (x: number, y: number, width: number, height: number, color: string) => {
    for (let row = y; row < y + height; row++) for (let col = x; col < x + width; col++) pixels[row][col] = color;
  };
  const stamp = (x: number, y: number, rows: readonly string[]) => rows.forEach((row, dy) => [...row].forEach((color, dx) => {
    if (color !== '.') pixels[y + dy][x + dx] = color;
  }));

  if (pose === 'back') {
    // ブーツはマントより先に描き、白い裾から先端だけを出す。
    rect(12, 26, 4, 5, '0'); rect(18, 26, 4, 5, '0');
    rect(13, 28, 2, 2, 'd'); rect(19, 28, 2, 2, 'd');
    rect(12, 30, 3, 1, 'e'); rect(18, 30, 3, 1, 'e');
    stamp(6, 12, [
      '......00000000......',
      '....009999999900....',
      '...09989999998990...',
      '..0922999999992290..',
      '..0922222222222290..',
      '..0912333333322190..',
      '.092123333333321290.',
      '.092123333333321290.',
      '.092123333333321290.',
      '.092112333333211290.',
      '.092112333333211290.',
      '09221123333332112290',
      '09221123333332112290',
      '09221123333332112290',
      '09221123333332112290',
      '09999999999999999990',
      '.098999999999999890.',
      '..0000000000000000..',
    ]);
    // 縁は前面と同じ白＋黒い斑点。縦の赤い影は布の折り目。
    for (const [x, y] of [[9, 16], [8, 21], [7, 25], [24, 16], [25, 21], [26, 25], [11, 28], [20, 28]]) rect(x, y, 1, 1, '0');
    // 後頭部は白髪をまとまった面にし、顔や目を描かない。
    stamp(10, 6, [
      '..00000000..',
      '.0789999870.',
      '078999999870',
      '078999999870',
      '078998999870',
      '.0788898870.',
      '..0788870...',
      '...00000....',
    ]);
    stamp(11, 1, [
      '....00....',
      '...0560...',
      '00.0560.00',
      '0605665060',
      '0655555560',
      '0456556540',
      '.05555550.',
    ]);
  } else {
    // 横向きはマントを背中側（左）へ寄せ、顔と足先を右へ向ける。
    rect(13, 26, 5, 5, '0'); rect(19, 26, 4, 5, '0');
    rect(14, 28, 3, 2, 'd'); rect(20, 28, 2, 2, 'd');
    rect(14, 30, 4, 1, 'e'); rect(20, 30, 3, 1, 'e');
    rect(14, 27, 3, 1, '5'); rect(20, 27, 2, 1, '5');
    stamp(7, 13, [
      '....0000000...',
      '...092211190..',
      '...092222190..',
      '..0923322190..',
      '..0923332190..',
      '..0923332190..',
      '..0923332190..',
      '.09223332190..',
      '.09223332190..',
      '.09223332190..',
      '.09223332190..',
      '092223332190..',
      '092223332190..',
      '092222222190..',
      '099999999990..',
      '.0999999990...',
      '..00000000....',
    ]);
    // 生成した正面と同じクリーム色の服、金の帯と赤い留め石。
    rect(17, 15, 5, 11, '0'); rect(18, 16, 3, 9, 'f');
    rect(18, 17, 1, 7, '9'); rect(21, 17, 1, 8, '5');
    rect(18, 21, 4, 1, '4'); rect(18, 22, 4, 1, '6'); rect(20, 22, 1, 1, '2');
    rect(18, 25, 4, 1, '6');
    rect(20, 17, 3, 4, '0'); rect(21, 17, 1, 3, 'f');
    rect(20, 20, 4, 3, '0'); rect(21, 21, 2, 1, 'b'); rect(21, 20, 1, 1, 'c');
    stamp(10, 12, ['..000000..', '.09999990.', '0999899900', '.09899990.']);
    for (const [x, y] of [[11, 15], [9, 20], [8, 25], [11, 28], [16, 28]]) rect(x, y, 1, 1, '0');
    // 白髪→肌→鼻→短い白ひげ。黒目は1×2ドット。
    stamp(11, 6, [
      '..0000000...',
      '.078998cb0..',
      '0789998cc0..',
      '078998bc0c0.',
      '078998bccbc0',
      '.07898bcc00.',
      '..078998980.',
      '...0799990..',
      '....07880...',
      '.....000....',
    ]);
    rect(19, 9, 1, 2, '0'); rect(19, 8, 1, 1, '7');
    stamp(12, 1, [
      '....00....',
      '...0560...',
      '.0.0560.0.',
      '0605665060',
      '0655555560',
      '0455552240',
      '.05555650.',
    ]);
  }
  return { width: 32, height: 32, palette, pixels: pixels.map(row => row.join('')) };
}

const right = paint('right');
export const kingDirectionPixels = {
  back: paint('back'),
  left: { ...right, pixels: right.pixels.map(row => [...row].reverse().join('')) },
  right,
} as const;
